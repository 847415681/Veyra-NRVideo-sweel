cmake_minimum_required(VERSION 3.24)
project(VeyraBilibiliTests LANGUAGES CXX)
enable_testing()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Network Test Qml Quick QuickControls2)
include(../../cmake/VeyraBilibili.cmake)
add_executable(veyra_bilibili_tests PlaybackTests.cpp)
set_target_properties(veyra_bilibili_tests PROPERTIES AUTOMOC ON)
target_link_libraries(veyra_bilibili_tests PRIVATE veyra_bilibili Qt6::Test)
add_test(NAME bilibili COMMAND veyra_bilibili_tests)
add_executable(veyra_bilibili_qml_tests QmlTests.cpp)
set_target_properties(veyra_bilibili_qml_tests PROPERTIES AUTOMOC ON)
target_link_libraries(veyra_bilibili_qml_tests PRIVATE veyra_bilibili Qt6::Test Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(veyra_bilibili_qml_tests PRIVATE VEYRA_QML_ROOT="${VEYRA_BILIBILI_ROOT}/qml")
add_test(NAME bilibili_qml COMMAND veyra_bilibili_qml_tests)
add_executable(veyra_bilibili_probe Probe.cpp)
target_link_libraries(veyra_bilibili_probe PRIVATE veyra_bilibili Qt6::Qml Qt6::Quick Qt6::QuickControls2)
if(MSVC)
  target_compile_options(veyra_bilibili_tests PRIVATE /utf-8)
  target_compile_options(veyra_bilibili_probe PRIVATE /utf-8)
  target_compile_options(veyra_bilibili_qml_tests PRIVATE /utf-8)
endif()
